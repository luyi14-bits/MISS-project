```python
    def build(self, session_id: str, user_message: str, profile: MISSProfile) -> dict:
```

```python
class LLMCaller:
    async def call(self, messages: list[dict], model_config: dict) -> dict:
        # 非流式模式：
        response = await openai.chat.completions.create(
            model=model_config.get("model", "gpt-4o"),
            messages=messages,
            temperature=model_config.get("temperature", 1.0),
            top_p=model_config.get("top_p", 0.92),
            max_tokens=model_config.get("max_tokens", 1024),
            frequency_penalty=model_config.get("frequency_penalty", 0.1),
            response_format={"type": "json_object"},
        )
        return self._parse_json_response(response.choices[0].message.content)

    def _parse_json_response(self, raw: str) -> dict:
        # 三级容错：严格json → regex提取 → 降级spoken=raw
        try: return json.loads(raw)
        except:
            m = re.search(r'\{[\s\S]*"spoken"[\s\S]*\}', raw)
            if m: return json.loads(m.group())
            return {"inner_thought": "", "spoken": raw}
```
**输入**：messages 列表 + 模型配置
**输出**：`{"inner_thought": str, "spoken": str}` 解析后的双轨响应

### Task 4.2 ★ 记忆重要性评分 + 摘要生成
**目标**：在消息移出工作窗口时，自动评估重要性并分级存储
**设计约束**：本项目仅使用一个 LLM API（用于主对话），**不为记忆评分分配额外的 API 额度**。评分和摘要均使用本地规则引擎实现，不依赖外部模型调用。

**流程**：
```
ConversationStore.get_overflow_messages()
    │
    ▼
MemoryScorer.score(overflow_messages)     ← 纯本地规则评分（无API调用）
    │  基于关键词权重 + 消息长度 + 角色模式 + 情感词密度
    │  返回 [{content, importance(0-100), category(event/fact/emotional)}, ...]
    │
    ▼
MemorySummarizer.process(session_id, scored_results)
    │  importance >= 80 → 保留原文 → 写入 MemoryEntry 表
    │  40 <= importance < 80 → 截取首句作为摘要（≤50字）→ 写入 MemoryEntry 表
    │  importance < 40 → 丢弃
    │
    ▼
MemoryEntry 表 (session_id, content, importance, category, timestamp)
```

**核心数据结构**：
```python
class MemoryEntry(BaseModel):
    session_id: str
    content: str          # 原始或摘要
    importance: int       # 0-100
    timestamp: datetime
    embedding: list[float] | None  # Phase 4.3 使用
    category: str         # "event" / "fact" / "emotional"
```

**MemoryScorer — 规则评分引擎（纯本地，零API调用）**：
```python
class MemoryScorer:
    def score(self, overflow_messages: list[dict]) -> list[dict]:
        """纯规则评分，不调用任何外部API"""
        results = []
        for msg in overflow_messages:
            content = msg.get("content", "")
            role = msg.get("role", "user")

            # 1. 长度分 (0-30)：越长信息量越大
            length_score = min(len(content) / 10, 30)

            # 2. 关键词分 (0-40)：检测个人信息、决策、情感关键词
            personal_keywords = ["我叫", "我是", "我喜欢", "我讨厌", "我决定",
                "我的", "年龄", "生日", "家在", "工作", "电话", "地址"]
            decision_keywords = ["决定", "打算", "想要", "一定要", "必须", "选择"]
            emotional_keywords = ["开心", "难过", "生气", "害怕", "担心", "爱",
                "恨", "感动", "失望", "兴奋", "焦虑", "幸福"]
            kw_score = 0
            for kw in personal_keywords + decision_keywords:
                if kw in content:
                    kw_score += 10
            for kw in emotional_keywords:
                if kw in content:
                    kw_score += 5
            kw_score = min(kw_score, 40)

            # 3. 角色分 (0-15)：用户关于自身的信息更重要
            role_score = 15 if role == "user" else 5

            # 4. 情感密度分 (0-15)：感叹号、问号、语气词密度
            emotional_density = (content.count("！") + content.count("!") +
                content.count("？") + content.count("?") + content.count("～"))
            density_score = min(emotional_density * 3, 15)

            importance = min(int(length_score + kw_score + role_score + density_score), 100)

            # 分类
            category = self._classify(content)

            results.append({
                "content": content,
                "importance": importance,
                "category": category,
                "role": role,
                "id": msg.get("id"),
            })
        return results

    def _classify(self, content: str) -> str:
        emotional_words = ["开心","难过","生气","害怕","担心","爱","恨","感动",
            "失望","兴奋","焦虑","幸福","喜欢","讨厌","想","烦"]
        event_verbs = ["去","做","发生","买了","到了","去了","吃了","看了",
            "决定了","完成","开始"]
        emotional_count = sum(1 for w in emotional_words if w in content)
        event_count = sum(1 for v in event_verbs if v in content)
        if emotional_count > event_count:
            return "emotional"
        elif event_count > 0:
            return "event"
        else:
            return "fact"
```

**MemorySummarizer — 本地摘要生成（纯本地，零API调用）**：
```python
class MemorySummarizer:
    def process(self, session_id: str, scored_results: list[dict]) -> dict:
        retained = 0
        summarized = 0
        discarded = 0
        for item in scored_results:
            importance = item["importance"]
            content = item["content"]
            category = item["category"]
            if importance >= 80:
                self._save_memory(session_id, content, importance, category)
                retained += 1
            elif importance >= 40:
                summary = self._extract_summary(content)  # 本地截取，无API
                self._save_memory(session_id, summary, importance, category)
                summarized += 1
            else:
                discarded += 1
        return {"retained": retained, "summarized": summarized, "discarded": discarded}

    def _extract_summary(self, content: str) -> str:
        """取首句或前50字作为摘要"""
        if len(content) <= 50:
            return content
        # 尝试在第一个句号/感叹号/问号处截断
        for sep in ["。", "！", "？", ".", "!", "?", "\n"]:
            idx = content.find(sep)
            if 10 <= idx <= 50:
                return content[:idx + 1]
        return content[:50] + "..."
```

**依赖**：Task 4.1
**验收**：
- 长时间对话后自动触发评分与压缩
- 个人关键词消息得分 > 60
- 纯寒暄消息（如"你好"、"嗯"）得分 < 30
- 所有评分/摘要过程不产生任何外部 API 调用
